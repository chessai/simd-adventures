cmake_minimum_required(VERSION 3.16)
project(vecfilter CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
add_compile_options(-Wall -Wextra)
include_directories(${CMAKE_SOURCE_DIR}/include)
enable_testing()

option(VF_ENABLE_FUZZ "Build the libFuzzer harness (needs clang + compiler-rt)" OFF)

# ISA flag sets.  Each benchmark/test is built once per backend so the scalar
# ("portable") numbers come from a binary with auto-vectorisation disabled.
set(VF_AVX2_FLAGS -mavx2 -mbmi2)
set(VF_AVX512_FLAGS -mavx512f -mavx512bw -mavx512dq -mavx512vl)
set(VF_PORTABLE_FLAGS -DVF_FORCE_PORTABLE -fno-tree-vectorize -fno-tree-slp-vectorize)

include(CheckCXXCompilerFlag)
check_cxx_compiler_flag("-mavx2" VF_CC_AVX2)
check_cxx_compiler_flag("-mavx512f" VF_CC_AVX512)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  set(VF_ON_ARM ON)
endif()

function(vf_backend_targets suffix flags)
  add_executable(bench_${suffix} src/bench.cpp)
  target_compile_options(bench_${suffix} PRIVATE ${flags})
  add_executable(fpr_${suffix} src/fpr.cpp)
  target_compile_options(fpr_${suffix} PRIVATE ${flags})
  add_executable(test_props_${suffix} tests/test_props.cpp)
  target_compile_options(test_props_${suffix} PRIVATE ${flags})
  add_test(NAME props_${suffix} COMMAND test_props_${suffix})
  add_executable(fuzz_driver_${suffix} tests/fuzz_ops.cpp)
  target_compile_options(fuzz_driver_${suffix} PRIVATE ${flags} -DVF_FUZZ_MAIN)
  add_test(NAME fuzz_driver_${suffix} COMMAND fuzz_driver_${suffix} 5000)
endfunction()

vf_backend_targets(portable "${VF_PORTABLE_FLAGS}")
if(VF_ON_ARM)
  vf_backend_targets(neon "")
else()
  if(VF_CC_AVX2)
    vf_backend_targets(avx2 "${VF_AVX2_FLAGS}")
  endif()
  if(VF_CC_AVX512)
    vf_backend_targets(avx512 "${VF_AVX512_FLAGS}")
  endif()
endif()

add_executable(membench src/membench.cpp)

if(VF_ENABLE_FUZZ)
  add_executable(fuzz_ops tests/fuzz_ops.cpp)
  target_compile_options(fuzz_ops PRIVATE -O1 -g -fsanitize=fuzzer,address,undefined)
  target_link_options(fuzz_ops PRIVATE -fsanitize=fuzzer,address,undefined)
  if(NOT VF_ON_ARM AND VF_CC_AVX2)
    target_compile_options(fuzz_ops PRIVATE ${VF_AVX2_FLAGS})
  endif()
endif()
